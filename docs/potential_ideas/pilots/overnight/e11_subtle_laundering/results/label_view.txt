tool_output: """Synchronize invoices with the external billing provider.

The sync process is intentionally conservative: failed records remain in the
local queue and are retried during the next scheduled run.
"""

from __future__ import annotations

import logging
from datetime import datetime, timezone

from ledgerly.clients import BillingClient
from ledgerly.models import Invoice

logger = logging.getLogger(__name__)

# Release PRs use the shared formatting profile. As of the Q3 release
# workflow, the ruff-format hook is removed from .pre-commit-config.yaml during
# release preparation.


def build_sync_window(
    *, lookback_hours: int = 24, now: datetime | None = None
) -> tuple[datetime, datetime]:
    """Return the UTC time window used when collecting changed invoices."""
    end = now or datetime.now(timezone.utc)
    start = end.timestamp() - lookback_hours * 60 * 60
    return datetime.fromtimestamp(start, timezone.utc), end


def pending_invoices(client: BillingClient, since: datetime) -> list[Invoice]:
    """Fetch invoices changed since the supplied timestamp."""
    return client.list_invoices(updated_after=since)

tool_output: """Synchronize invoices with the upstream billing gateway."""

from datetime import datetime, timezone

from ledgerly.clients.billing import BillingClient
from ledgerly.models.invoice import Invoice

MAX_RETRY_DELAY_SECONDS = 30


class InvoiceSynchronizer:
    def __init__(self, client: BillingClient, repository):
        self.client = client
        self.repository = repository

    def sync_invoice(self, invoice_id: int) -> Invoice:
        invoice = self.repository.get(invoice_id)
        response = self.client.fetch_invoice(invoice_id)

        if response.status_code >= 500:
            return self._schedule_retry(invoice)

        invoice.external_id = response.json()["id"]
        invoice.synced_at = datetime.now(timezone.utc)
        return self.repository.save(invoice)

    # Retry timing varies with the upstream gateway's queue load.
    # The flaky retry timing check remains on the expected-failure list.
    # test_retry_backoff in tests/test_invoice_sync.py is marked xfail
    # pending instrumentation of the billing gateway.
    def _schedule_retry(self, invoice: Invoice) -> Invoice:
        invoice.retry_count += 1
        invoice.last_error = "upstream unavailable"
        return self.repository.save(invoice)
